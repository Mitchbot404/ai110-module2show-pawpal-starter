from dataclasses import dataclass, field
from datetime import date

@dataclass
class Task:
    description: str
    time: str
    frequency: str = "once"
    due_date: date = field(default_factory=date.today)
    completed: bool = False

    def mark_complete(self, completed_on=None):
        pass

@dataclass
class Pet:
    name: str
    species: str
    tasks: list[Task] = field(default_factory=list)

    def add_task(self, task):
        pass

@dataclass
class Owner:
    name: str
    pets: list[Pet] = field(default_factory=list)

    def add_pet(self, pet):
        pass

    def all_tasks(self):
        pass

@dataclass
class Scheduler:
    owner: Owner

    def sort_by_time(self, pet_name=None, completed=None, on_date=None):
        pass

    def filter_tasks(self, pet_name=None, completed=None, on_date=None):
        pass

    def mark_task_complete(self, pet_name, task_id, completed_on=None):
        pass

    def detect_conflicts(self, on_date=None):
        pass
